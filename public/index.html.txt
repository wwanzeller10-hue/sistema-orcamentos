    <script>
        const CUSTOS_BASE = { malharia: 20.00, comunicacao_visual: 25.00, grafica: 0.25 };

        function alternarCampos() {
            const cat = document.getElementById('categoria').value;
            const boxLargura = document.getElementById('boxLargura');
            const boxAltura = document.getElementById('boxAltura');
            if (cat === 'comunicacao_visual') {
                boxLargura.classList.remove('hidden');
                boxAltura.classList.remove('hidden');
            } else {
                boxLargura.classList.add('hidden');
                boxAltura.classList.add('hidden');
            }
            recalcular();
        }

        function recalcular() {
            const cat = document.getElementById('categoria').value;
            const qtd = parseFloat(document.getElementById('quantidade').value) || 1;
            let custoMateriais = 0;

            if (cat === 'comunicacao_visual') {
                const larg = parseFloat(document.getElementById('largura').value) || 0;
                const alt = parseFloat(document.getElementById('altura').value) || 0;
                custoMateriais = (larg * alt * qtd) * CUSTOS_BASE.comunicacao_visual;
            } else {
                custoMateriais = qtd * CUSTOS_BASE[cat];
            }

            const precoSugerido = custoMateriais * 2.0;
            document.getElementById('txtCustoMateriais').innerText = `R$ ${custoMateriais.toFixed(2)}`;
            document.getElementById('precoCobrado').value = precoSugerido.toFixed(2);
        }

        let timerBusca;
        function buscarSimilares() {
            clearTimeout(timerBusca);
            timerBusca = setTimeout(async () => {
                const cat = document.getElementById('categoria').value;
                const desc = document.getElementById('descricao').value;
                if (desc.length < 3) {
                    document.getElementById('painelSimilares').classList.add('hidden');
                    return;
                }
                const res = await fetch(`/api/orcamentos/similares?categoria=${cat}&descricao=${encodeURIComponent(desc)}`);
                const dados = await res.json();
                const painel = document.getElementById('painelSimilares');
                const lista = document.getElementById('listaSimilares');
                lista.innerHTML = '';

                if (dados.length > 0) {
                    painel.classList.remove('hidden');
                    dados.forEach(item => {
                        const dataFmt = new Date(item.data).toLocaleDateString('pt-BR');
                        const div = document.createElement('div');
                        div.className = 'flex justify-between items-center bg-white p-2 rounded border border-amber-200 cursor-pointer hover:bg-amber-100 transition';
                        div.innerHTML = `<span><strong>${item.cliente}</strong> (${dataFmt}) — ${item.quantidade} un. de "${item.descricao}"</span><span class="font-bold text-amber-900">Cobrado: R$ ${parseFloat(item.preco_cobrado).toFixed(2)}</span>`;
                        div.onclick = () => { document.getElementById('precoCobrado').value = parseFloat(item.preco_cobrado).toFixed(2); };
                        lista.appendChild(div);
                    });
                } else {
                    painel.classList.add('hidden');
                }
            }, 350);
        }

        document.getElementById('formOrcamento').addEventListener('submit', async (e) => {
            e.preventDefault();
            const dados = {
                cliente: document.getElementById('cliente').value,
                categoria: document.getElementById('categoria').value,
                descricao: document.getElementById('descricao').value,
                quantidade: parseInt(document.getElementById('quantidade').value),
                largura: parseFloat(document.getElementById('largura').value) || null,
                altura: parseFloat(document.getElementById('altura').value) || null,
                custo_materiais: parseFloat(document.getElementById('txtCustoMateriais').innerText.replace('R$', '').trim()),
                preco_cobrado: parseFloat(document.getElementById('precoCobrado').value)
            };
            const res = await fetch('/api/orcamentos', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(dados)
            });
            if (res.ok) {
                alert('✅ Orçamento gravado no banco de dados com sucesso!');
                document.getElementById('painelSimilares').classList.add('hidden');
            }
        });

        function gerarPropostaPDF() {
            document.getElementById('pdfCliente').innerText = document.getElementById('cliente').value || 'Cliente Balcão';
            document.getElementById('pdfDescricao').innerText = document.getElementById('descricao').value;
            document.getElementById('pdfQtd').innerText = document.getElementById('quantidade').value;
            document.getElementById('pdfValor').innerText = 'R$ ' + parseFloat(document.getElementById('precoCobrado').value).toFixed(2);
            document.getElementById('pdfData').innerText = 'Data: ' + new Date().toLocaleDateString('pt-BR');

            const elemento = document.getElementById('modeloPDF');
            html2pdf().set({
                margin: 10,
                filename: `Orcamento_${document.getElementById('cliente').value || 'Balcao'}.pdf`,
                image: { type: 'jpeg', quality: 0.98 },
                html2canvas: { scale: 2 },
                jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' }
            }).from(elemento).save();
        }

        function enviarWhatsApp() {
            const cliente = document.getElementById('cliente').value;
            const desc = document.getElementById('descricao').value;
            const qtd = document.getElementById('quantidade').value;
            const preco = document.getElementById('precoCobrado').value;
            const mensagem = `Olá *${cliente}*, tudo bem?\n\nSegue o orçamento referente ao seu pedido:\n• *Produto:* ${desc}\n• *Quantidade:* ${qtd}\n• *Valor Total:* R$ ${preco}\n\nFicamos no aguardo para aprovação e início da produção!`;
            window.open(`https://api.whatsapp.com/send?text=${encodeURIComponent(mensagem)}`, '_blank');
        }

        alternarCampos();
